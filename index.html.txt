  <script>
    // Вставь сюда ссылку на data.json с GitHub Pages
    const DATA_URL = "https://pybjib.github.io/train-data/data.json";

    async function loadTrains() {
      const station = document.getElementById('stationSelect').value;
      const container = document.getElementById('results');
      container.innerHTML = 'Загрузка...';

      try {
        const res = await fetch(DATA_URL);
        if (!res.ok) throw new Error('Не удалось загрузить данные');
        const data = await res.json();

        const trains = data[station] || [];
        container.innerHTML = '';

        if (trains.length === 0) {
          container.innerHTML = '<p>Нет данных для этой станции.</p>';
          return;
        }

        trains.forEach(t => {
          const platformText = t.platform
            ? t.platform
            : '<span class="platform-unknown">Путь пока неизвестен</span>';

          const card = document.createElement('div');
          card.className = 'card';
          card.innerHTML = `
            <div><span class="label">Поезд</span><span class="value">${t.number}</span></div>
            <div><span class="label">Маршрут</span><span class="value">${t.title}</span></div>
            <div><span class="label">Прибытие</span><span class="value">${t.arrival}</span></div>
            <div><span class="label">Путь</span><span class="value">${platformText}</span></div>
          `;
          container.appendChild(card);
        });
      } catch (e) {
        console.error(e);
        container.innerHTML = '<p>Ошибка загрузки данных. Попробуйте позже.</p>';
      }
    }
  </script>

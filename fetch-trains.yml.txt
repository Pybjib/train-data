name: Fetch train data

on:
  schedule:
    - cron: "*/15 * * * *"  # каждые 15 минут
  workflow_dispatch:       # можно запустить вручную кнопкой в Actions

jobs:
  fetch:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: Install dependencies
        run: pip install httpx

      - name: Fetch data from Yandex Rasp
        env:
          YANDEX_API_KEY: ${{ secrets.YANDEX_API_KEY }}
        run: python ./.github/scripts/fetch_data.py

      - name: Commit and push
        run: |
          git config --local user.email "action@github.com"
          git config --local user.name "GitHub Action"
          git add data.json
          git commit -m "Update train data" || true
          git push
